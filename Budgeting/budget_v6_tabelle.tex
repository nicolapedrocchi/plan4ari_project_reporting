\section*{Convenzioni}
\begin{itemize}
  \item Importi in euro. Colonne: \textbf{FR} = Fundamental Research (ricerca fondamentale), \textbf{IR} = Industrial Research (ricerca industriale), \textbf{ED} = Experimental Development (sviluppo sperimentale).
  \item Soggetti: \textbf{HI} = Host Institution (COMAU, azienda ospitante), \textbf{AI} = Affiliation Institution, \textbf{RO} = Research Organization (CNR). Nel foglio ``SOTTOMESSO'' gli stessi soggetti sono chiamati rispettivamente Host Organization, Affiliation Institution e Research Institution.
  \item Voci di costo: \emph{direct cost} (costi diretti diversi dal personale: viaggi, materiali, robot, sistemi di visione, PC, sensori, consumabili, disseminazione), \emph{structured personnel} (personale strutturato), \emph{personnel to be acquired} (personale da acquisire), \emph{indirect costs} (spese generali).
  \item Con $c^{s}_{v,k}$ si indica il costo del soggetto $s\in\{\mathrm{HI},\mathrm{AI},\mathrm{RO}\}$, voce $v$, tipologia di attivit\`a $k\in\{\FR,\IR,\ED\}$; con $a^{s}_{v,k}$ la corrispondente agevolazione.
  \item I valori riportati sono quelli calcolati e salvati nel file Excel; le formule descritte nei preamboli sono quelle presenti nelle celle.
\end{itemize}

\subsection{Budget totale sottomesso}
\textbf{Formule e assunzioni.}
\begin{itemize}
  \item Il personale della RO (Research Institution) e della AI \`e inserito come valore fisso. Il personale della HI \`e calcolato come mesi-persona per una tariffa mensile di 5\,500\,\euro:
  \[
    c^{\mathrm{HI}}_{\mathrm{pers},k} = \mathrm{PM}^{\mathrm{HI}}_k \cdot 5\,500, \qquad \mathrm{PM}^{\mathrm{HI}} = (25,\ 410,\ 105)\ \text{per } (\FR,\IR,\ED),\ \text{in totale } 540\ \mathrm{PM}.
  \]
  \item Direct other costs: valori fissi. Indirect costs forfettari al 25\,\% di personale pi\`u costi diretti:
  \[ c_{\mathrm{ind},k} = 0.25\,\bigl(c_{\mathrm{pers},k} + c_{\mathrm{dir},k}\bigr). \]
  \item Costo totale per soggetto: $c_{\mathrm{tot},k} = c_{\mathrm{pers},k} + c_{\mathrm{dir},k} + c_{\mathrm{ind},k}$; il totale di riga \`e $\sum_k c_{\cdot,k}$.
  \item Calcoli di controllo a lato della tabella: per la RO i mesi-persona sono stimati con una tariffa di 4\,500\,\euro/mese ($\mathrm{PM}^{\mathrm{RO}}_k = c^{\mathrm{RO}}_{\mathrm{pers},k}/4\,500$, cio\`e 42.2, 82.2, 7.8, totale 132.2 PM); sono inoltre riportate le quote di personale sul totale ($E_9/E_4 = 74.4\,\%$ per la RO, $E_{11}/E_6 = 73.4\,\%$ per la HI), le quote di costi diretti ($5.6\,\%$ e $6.6\,\%$) e il valore dell'80\,\% dei costi della HI (3\,235\,000\,\euro).
\end{itemize}

\subsection{Quota cofinanziata}
\textbf{Formule e assunzioni.}
\begin{itemize}
  \item Quota di cofinanziamento: $1-\rho_k$, cio\`e $0\,\%$, $35\,\%$, $60\,\%$.

\end{itemize}

  e analogamente per il personale da acquisire.
  \item Le righe ``Tasso'' sono i tassi effettivi $a^{s}_{\mathrm{tot},k}/c^{s}_{\mathrm{tot},k}$.
  \item L'ultimo blocco (nel file etichettato ``Research Organization costs'', ma in realt\`a \`e AI+RO) somma voce per voce le agevolazioni di AI e RO; il totale coincide con i 809\,140.40\,\euro{} dell'Allegato~C (controlli a lato uguali a 0).
\end{itemize}
